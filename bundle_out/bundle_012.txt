void fsm_init_gates() {
    for (size_t i = 0; i < sizeof(g_fsm_gate_flags) / sizeof(g_fsm_gate_flags[0]); ++i) {
        g_fsm_gate_flags[i] = false;
    }
    g_fsm_uart_only_mode = false;
    g_fsm_uart_only_reason[0] = '\0';

void fsm_set_gate_flag(fsm_gate_t gate, bool value) {
    if (gate <= FSM_GATE_SELFTEST) {
        g_fsm_gate_flags[gate] = value;
    }
}

bool fsm_get_gate_flag(fsm_gate_t gate) {
    if (gate <= FSM_GATE_SELFTEST) {
        return g_fsm_gate_flags[gate];
    }
    return false;
}

bool fsm_all_gates_ok() {
    for (size_t i = 0; i < sizeof(g_fsm_gate_flags) / sizeof(g_fsm_gate_flags[0]); ++i) {
        if (!g_fsm_gate_flags[i]) {
            return false;
        }
    }
    return true;
}

    }

  if (g_displaySleeping) return;

  }

  LOG_INIT_BEGIN("Runtime-Tasks");

  g_lastUserActivityMs = now_ms();
  pm_wake_if_needed();

  g_sysTicks = 0;
  g_pmTicks  = 0;

  setTickBudgetMs(5);

  g_taskIdBOOT = addTaskImmediateNamed("boot.guard", BOOT_GUARD_TICK_MS, task_boot_budget_guard, true);

  start_power_manager_task(200);

  addTaskPhasedNamed("tel.housekeeping", 2000, TELEMETRY_SD_STALE_MS / 2, task_telemetry_housekeeping, true);

static constexpr unsigned long WDT_INTERVAL_MS = 2000;

void watchdog_feed(void) { last_alive = millis(); }

void watchdog_safe_delay(uint32_t ms) {
    uint32_t start = millis();
    while (millis() - start < ms) {
        watchdog_feed();
        delay(1);
        yield();
    }
}

static void safe_shutdown_once() {
  static bool done = false;
  if (done) return;
  done = true;
  set_backlight_percent(0);
  display_sleep();
}

static void watchdogTask(void*) {
  for (;;) {
    unsigned long now  = millis();
    unsigned long last = last_alive;
    const unsigned long timeout_ms = platform.wdt_ui_ms;
    if (now - last > timeout_ms) {
      log_flush("wdt_timeout");
      LOGE("Watchdog: timeout now=%lu last=%lu limit=%lu -> restart", now, last, timeout_ms);
      safe_shutdown_once();
      vTaskDelay(pdMS_TO_TICKS(100));
      esp_restart();
    }
    vTaskDelay(pdMS_TO_TICKS(WDT_INTERVAL_MS));
  }
}

static void _wdt_selftest_tick() {
  eTaskState st = eTaskGetState(g_wdtHandle);
  if (st == eRunning || st == eReady || st == eBlocked) {
    LOG_INIT_READY("Watchdog");
  } else {
    LOG_INIT_FAIL("Watchdog", "selftest state=%d", (int)st);
  }
}

void watchdog_init() {
  LOG_INIT_BEGIN("Watchdog");
  if (g_wdtHandle) {
    LOGW("Watchdog: bereits gestartet");
    return;
  }
  last_alive = millis();
  BaseType_t rc = xTaskCreatePinnedToCore(watchdogTask, "wdt", 2048, nullptr, 1, &g_wdtHandle, 0);
  if (rc != pdPASS || g_wdtHandle == nullptr) {
    LOG_INIT_FAIL("Watchdog", "task create rc=%ld", (long)rc);
    return;
  }
  (void)addTaskPhasedNamed("wdt.selftest", 50, 0, _wdt_selftest_tick, false);
}


void tel_render_on_flush(uint32_t ms, uint32_t px, bool full_flush) {
  g_render.ms_sum += ms;
  if (ms > g_render.ms_max) g_render.ms_max = ms;
  g_render.px_sum += px;
  g_render.cnt++;
  const bool eff_full = full_flush || (g_px_full && px >= g_px_full);
  if (eff_full) g_render.full_cnt++;
}

void telemetry_get_render(uint32_t* ms_avg, uint32_t* ms_max, uint32_t* px_avg, uint8_t*  fill_pct, uint32_t* full_flush_cnt) {
  const uint32_t cnt = g_render.cnt ? g_render.cnt : 1;
  if (ms_avg)  *ms_avg  = (uint32_t)(g_render.ms_sum / cnt);
  if (ms_max)  *ms_max  = g_render.ms_max;
  if (px_avg)  *px_avg  = (uint32_t)(g_render.px_sum / cnt);
  if (fill_pct) {
    uint32_t avg_px = (uint32_t)(g_render.px_sum / cnt);
    uint32_t pct = (g_px_full ? (avg_px * 100u) / g_px_full : 0u);
    if (pct > 100u) pct = 100u;
    *fill_pct = (uint8_t)pct;
  }
  if (full_flush_cnt) *full_flush_cnt = g_render.full_cnt;
}

  if (largest_k) *largest_k = (uint32_t)(( (larg8>largps?larg8:largps) )/1024u);
#else
  if (free_k)      *free_k = 0;
  if (min_free_k)  *min_free_k = 0;
  if (largest_k)   *largest_k = 0;
#endif
}

bool telemetry_sd_is_stale() {
    sd_status_cache_t cache;
    if (!sd_try_get_status_cache(&cache) || cache.last_successful_probe_ms == 0) {
        return true; 
    }
    return (now_ms() - cache.last_successful_probe_ms) > TELEMETRY_SD_STALE_MS;
}

bool telemetry_sd_ok() {
  sd_status_cache_t cache;
  if (!sd_try_get_status_cache(&cache)) {
      return false; 
  }
  return cache.mounted && !telemetry_sd_is_stale();
}

  sd_status_cache_t cache;
  if (telemetry_sd_ok() && sd_try_get_status_cache(&cache)) {

  return 0; 
}

int32_t telemetry_wdt_margin_ms() {
  const uint32_t timeout_ms = g_wdt_timeout_cfg_ms ? g_wdt_timeout_cfg_ms : platform.wdt_ui_ms;
  uint32_t now = now_ms();
  uint32_t last = (uint32_t)last_alive;
  uint32_t elapsed = now - last;
  if (elapsed >= timeout_ms) return (int32_t)(timeout_ms) - (int32_t)elapsed; 
  return (int32_t)(timeout_ms - elapsed);
}

struct ClockTargets {
  lv_obj_t* lbl_time{nullptr};
  lv_obj_t* lbl_date{nullptr};
  lv_obj_t* lbl_dotw{nullptr};
};
static ClockTargets  g_clk;
static datetime_t    g_last{};
static volatile bool g_dirty = false;
static const char* dotw_name(uint8_t d) {
  static const char* N[7] = {"So","Mo","Di","Mi","Do","Fr","Sa"};
  return (d < 7) ? N[d] : "?";
}
static void clock_apply_() {
  if (!g_dirty) return;
  g_dirty = false;
  const datetime_t now = g_last;
  if (!rtc_has_valid_time()) {
    if (g_clk.lbl_time)  lv_label_set_text_static(g_clk.lbl_time, "--:--:--");
    if (g_clk.lbl_date)  lv_label_set_text_static(g_clk.lbl_date, "--.--.----");
    if (g_clk.lbl_dotw)  lv_label_set_text_static(g_clk.lbl_dotw, "--");
    return;
  }
  if (g_clk.lbl_time) lv_label_set_text_fmt(g_clk.lbl_time, "%02u:%02u:%02u", now.hour, now.minute, now.second);
  if (g_clk.lbl_date) lv_label_set_text_fmt(g_clk.lbl_date, "%02u.%02u.%04u",  now.day,  now.month,  now.year);
  if (g_clk.lbl_dotw) lv_label_set_text_static(g_clk.lbl_dotw, dotw_name(now.dotw));
}
static void clock_timer_cb(lv_timer_t*) { clock_apply_(); }
static lv_timer_t* s_clock_tmr = nullptr;
static void rtc_hook_(const datetime_t* now) {
  g_last = *now;
  g_dirty = true;
}
void clock_bind(const ClockTargets& t) {
  g_clk = t;
  rtc_set_update_hook(&rtc_hook_);
  if (!s_clock_tmr) s_clock_tmr = lv_timer_create(clock_timer_cb, 100, nullptr);
  g_dirty = true;
  clock_apply_();
}
